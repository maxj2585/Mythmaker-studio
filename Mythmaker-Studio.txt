function relevantKnowledge(c, query, budget = 1400) {
  if (!c.knowledge) return "";
  const chunks = c.knowledge.split(/\n\s*\n/).flatMap((p) => p.match(/[\s\S]{1,500}(\s|$)/g) || []).map((x) => x.trim()).filter(Boolean);
  if (c.knowledge.length <= budget) return c.knowledge;
  const q = words(query);
  const scored = chunks.map((t, i) => ({ t, i, s: [...words(t)].filter((w) => q.has(w)).length })).sort((a, b) => b.s - a.s || a.i - b.i);
  let out = "";
  for (const { t } of scored) {
    if (out.length + t.length > budget) break;
    out += t + "\n";
  }
  return out.trim();
}

function speak(text, c) {
  if (!("speechSynthesis" in window)) return;
  speechSynthesis.cancel();
  const u = new SpeechSynthesisUtterance(text.replace(/\*[^*]*\*/g, "").replace(/\[[^\]]*\]/g, ""));
  u.pitch = c.voice?.pitch ?? 1;
  u.rate = c.voice?.rate ?? 1;
  const v = speechSynthesis.getVoices().find((x) => x.name === c.voice?.name);
  if (v) u.voice = v;
  speechSynthesis.speak(u);
}
function attachMic(btn, input) {
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  if (!SR) {
    btn.hidden = true;
    return;
  }
  btn.onclick = () => {
    const rec = new SR();
    rec.lang = navigator.language || "en-US";
    rec.interimResults = false;
    rec.onresult = (e) => {
      input.value += (input.value ? " " : "") + e.results[0][0].transcript;
      input.focus();
    };
    rec.onend = () => {
      btn.classList.remove("rec");
      btn.textContent = "\u{1F3A4}";
    };
    btn.classList.add("rec");
    btn.textContent = "\u25CF Listening";
    rec.start();
  };
}
async function rememberFrom(c, msgs) {
  if (c.memory === false || !hasWebGPU()) return;
  const transcript = msgs.slice(-10).map((m) => `${m.role === "user" ? userLabel() : c.name}: ${m.content}`).join("\n");
  try {
    const out = await withAI(async () => {
      const eng = await getEngine();
      const res = await eng.chat.completions.create({
        temperature: 0.2,
        max_tokens: 160,
        messages: [
          { role: "system", content: "You pick out facts worth remembering long term from a chat: names, preferences, plans, relationships, and story events. Reply with a JSON array of up to 3 short strings, or [] if nothing is worth keeping. JSON only." },
          { role: "user", content: transcript }
        ]
      });
      return res.choices[0].message.content;
    });
    const facts = JSON.parse(out.slice(out.indexOf("["), out.lastIndexOf("]") + 1));
    if (Array.isArray(facts)) addMemories(c.id, facts.filter((f) => typeof f === "string"));
  } catch (err) {
    console.warn("memory update skipped", err);
  }
}

async function shrinkImage(file) {
  if (!file) return "";
  const img = await createImageBitmap(file);
  const size = 256, canvas = document.createElement("canvas");
  canvas.width = canvas.height = size;
  const s = Math.min(img.width, img.height);
  canvas.getContext("2d").drawImage(img, (img.width - s) / 2, (img.height - s) / 2, s, s, 0, 0, size, size);
  return canvas.toDataURL("image/jpeg", 0.85);
}

        const named = chars.filter((c) => lower.includes(c.name.toLowerCase().split(" ")[0]));
        let speakers = [...named, ...chars.filter((c) => !named.includes(c) && Math.random() * 100 < c.freeWill)].slice(0, 2);
        if (!speakers.length) speakers = [chars[Math.floor(Math.random() * chars.length)]];
        for (const c of speakers) {
          if (!isHost()) break;
          await replyAs(c, m.content);
        }
      }
    } catch (err) {
      console.error(err);
      status.textContent = "The on-device model could not run. Try a smaller model in Settings, or reload.";
    } finally {
      busy = false;
    }
  }
  async function replyAs(c, query) {
    let content = "";
    await withAI(async () => {
      const eng = await getEngine((p, t) => {
        $view.querySelector("#bar").style.width = `${Math.round(p * 100)}%`;
        status.textContent = t;
      });
      status.textContent = `${c.name} is typing\u2026`;
      const history = msgs.slice(-20).map((m) => m.charId === c.id ? { role: "assistant", content: m.content } : { role: "user", content: `${m.name}: ${m.content}` });
      const peopleNames = [...new Set(Object.values(online).map((p) => p.name))];
      const res = await eng.chat.completions.create({
        temperature: Number(settings().temperature) || 0.8,
        max_tokens: 220,
        messages: [{ role: "system", content: systemPrompt(c, { group: chars, people: peopleNames, query }) }, ...history]
      });
      content = (res.choices[0].message.content || "").replace(new RegExp("^\\s*" + c.name.replace(/[.*+?^${}()|[\]\\]/g, "\\$&") + "\\s*:\\s*", "i"), "").trim();
    });
    if (content) await f.push(f.ref(f.db, `rooms/${code}/messages`), { kind: "char", charId: c.id, name: c.name, content: content.slice(0, 2e3), at: f.serverTimestamp() });
    showState();
  }
  hostBtn.onclick = async () => {
    await f.set(f.ref(f.db, `rooms/${code}/meta/host`), f.uid);
    preloadAI();
  };
  $view.querySelector("#invite").onclick = async () => {
    const url = `${location.origin}${location.pathname}#/room/${code}`;
    try {
      await navigator.clipboard.writeText(url);
      status.textContent = "Invite link copied. Anyone with the link can join.";
    } catch {
      prompt("Copy this invite link:", url);
    }
  };
  attachMic($view.querySelector("#mic"), input);
  input.addEventListener("keydown", (e) => {
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      $view.querySelector("#composer").requestSubmit();
    }
  });
  $view.querySelector("#composer").onsubmit = (e) => {
    e.preventDefault();
    const text = input.value.trim();
    if (!text) return;
    input.value = "";
    f.push(f.ref(f.db, `rooms/${code}/messages`), { kind: "user", uid: f.uid, name: me(), content: text.slice(0, 2e3), at: f.serverTimestamp() }).catch(() => {
      status.textContent = "Message not sent. Check your connection.";
    });
  };
  showState();
}
function route() {
  roomCleanup?.();
  if ("speechSynthesis" in window) speechSynthesis.cancel();
  const [, page, arg] = (location.hash || "#/").split("/");
  const id = arg ? decodeURIComponent(arg) : "";
  if (page === "create") editorView();
  else if (page === "edit") editorView(id);
  else if (page === "chat") chatView(id);
  else if (page === "import") importView(arg);
  else if (page === "settings") settingsView();
  else if (page === "groups") groupsView();
  else if (page === "group") groupChatView(id);
  else if (page === "rooms") roomsView();
  else if (page === "room") roomView(id);
  else galleryView();
  window.scrollTo(0, 0);
}
window.addEventListener("hashchange", route);
